"""Final-source qualification: every unit test, one owned pytest worker at a time."""
import hashlib,json,os,threading,time
from pathlib import Path
import psutil
from scripts import fast_test as runner,generate_metrics
root=Path.cwd(); folder=root/'docs/audits/evidence'
name='release_full_final_2026_10_12'
metadata=folder/(name+'.json'); resource_log=folder/(name+'_resources.log')
files=runner.discover_test_files(); batches=[files[i:i+50] for i in range(0,len(files),50)]
meta={'status':'running','owner_pid':os.getpid(),'started':time.time(),'workers':1,'math_threads':2,'input_sha256':generate_metrics.collection_fingerprint(),'test_files':len(files),'batch_size':50,'includes_slow':True,'results':[]}
def save(): metadata.write_text(json.dumps(meta,indent=2)+'\n',encoding='utf-8')
save(); stopped=threading.Event()
def sample():
 owner=psutil.Process()
 with resource_log.open('w',encoding='utf-8') as log:
  while not stopped.is_set():
   children=owner.children(recursive=True); rss=0
   for child in children:
    try: rss+=child.memory_info().rss
    except (psutil.NoSuchProcess,psutil.AccessDenied): pass
   log.write(json.dumps({'at':time.time(),'children':len(children),'owned_rss_bytes':rss,'available_bytes':psutil.virtual_memory().available})+'\n');log.flush()
   stopped.wait(10)
sampler=threading.Thread(target=sample,daemon=True);sampler.start()
try:
 for number,batch in enumerate(batches):
  meta['current_batch']=number;save()
  print(f'FULL batch {number+1}/{len(batches)}: {len(batch)} files',flush=True)
  args=[*[p.relative_to(root).as_posix() for p in batch],'--timeout=120','--tb=short','--continue-on-collection-errors']
  code,output=runner.run_pytest(args, max(900,len(batch)*25))
  receipt=folder/(name+f'_batch_{number:03d}.log');receipt.write_text(output,encoding='utf-8')
  counts=runner._parse_summary(output)
  row={'batch':number,'files':[p.relative_to(root).as_posix() for p in batch],'exit':code,'counts':counts,'measured':code in (0,1) and bool(counts),'failures':sorted({m.group(2) for m in runner._FAILED_RE.finditer(output)})}
  meta['results'].append(row);save();print(json.dumps({k:v for k,v in row.items() if k!='files'}),flush=True)
  assert generate_metrics.collection_fingerprint()==meta['input_sha256'],'Collection inputs changed during qualification'
 meta.update(status='complete',finished=time.time(),final_input_sha256=generate_metrics.collection_fingerprint())
 totals={}
 for row in meta['results']:
  for key,value in row['counts'].items(): totals[key]=totals.get(key,0)+value
 meta['totals']=totals
 meta['exit']=int(any(row['exit']!=0 or not row['measured'] for row in meta['results']))
 save(); print(json.dumps({'status':meta['status'],'totals':totals,'exit':meta['exit'],'elapsed_s':round(meta['finished']-meta['started'],1)}),flush=True)
finally:
 stopped.set();sampler.join(15)
raise SystemExit(meta['exit'])
