def run(command, name):
    started = time.monotonic()
    proc, tree = _start_process(command, env=env, memory_bytes=3*1024**3, cpu_rate=2000)
    buffers = [bytearray(), bytearray()]
    overflow = threading.Event()
    lock = threading.Lock()
    def drain(stream, index):
        while chunk := stream.read1(8192):
            with lock:
                if sum(map(len,buffers)) + len(chunk) > 128*1024:
                    overflow.set()
                    return
                buffers[index].extend(chunk)
            if index == 0:
                print(chunk.decode('utf-8','replace'),end='',flush=True)
    threads = [threading.Thread(target=drain,args=(stream,i),daemon=True)
               for i,stream in enumerate((proc.stdout,proc.stderr))]
    for thread in threads: thread.start()
    proc.stdin.close()
    code = 1
    failure = None
    try:
        while proc.poll() is None:
            if overflow.is_set(): raise RuntimeError('Soak output budget exceeded')
            if time.monotonic()-started > duration+180: raise TimeoutError('Soak lifetime budget exceeded')
            time.sleep(.25)
        code = proc.returncode
    except BaseException as exc:
        failure = repr(exc)
    finally:
        tree.close()
        proc.wait(timeout=5)
        for thread in threads: thread.join(5)
        for stream in (proc.stdout,proc.stderr): stream.close()
    assert not any(thread.is_alive() for thread in threads), 'Soak output readers did not close'
    output = buffers[0].decode('utf-8','replace')
    errors = buffers[1].decode('utf-8','replace')
    receipt = {'platform':name,'exit':code,'error':failure,'elapsed_s':round(time.monotonic()-started,1),
               'memory_limit_bytes':3*1024**3,'cpu_rate':2000,'tree_closed':tree.closed}
    log = json.dumps(receipt)+'\n'+output+'\n'+errors
    (root / f'docs/audits/evidence/release_{name}_soak_2026_10_12.log').write_text(log,encoding='utf-8')
    print(log,flush=True)
    if failure or overflow.is_set(): raise RuntimeError(failure or 'output overflow')
    return code
