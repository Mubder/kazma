"""Scan selected public receipts against configured secrets without emitting values."""
import json,re
from pathlib import Path
from dotenv import dotenv_values

root=Path.cwd(); values=set()
for path in (root/'.env',Path('C:/Users/balfa/kazma/.env')):
    if not path.is_file():continue
    for key,value in dotenv_values(path).items():
        if value and len(value)>=12 and any(part in key.upper() for part in ('KEY','TOKEN','SECRET','PASSWORD','DSN','DATABASE_URL')):
            values.add(value)
from release_select_evidence import selected_files
files=selected_files()
hits=[]
for path in files:
    text=path.read_text(encoding='utf-8',errors='replace')
    if any(value in text for value in values):hits.append(path.name)
result={'receipt_files_scanned':len(files),'configured_secret_matches':hits,'secret_values_emitted':False}
(root/'docs/audits/evidence/release_receipt_secret_scan_2026_10_12.log').write_text(json.dumps(result,indent=2)+'\n',encoding='utf-8')
print(json.dumps(result))
assert not hits,'Configured credentials appeared in the named receipts; do not publish them'
