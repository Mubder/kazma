from pathlib import Path
from types import SimpleNamespace
class Error(Exception): pass
core=Path(sys.argv[1]);old=Path(sys.argv[2])
def owner(source,root):
    tree=ast.parse(source.read_text(encoding='utf-8'))
    cls=next(n for n in tree.body if isinstance(n,ast.ClassDef) and n.name=='_PipelineJob')
    cls.body=[n for n in cls.body if isinstance(n,ast.FunctionDef) and n.name in {'__init__','checked','file_ref','restore','_read_artifact'}]
    space=dict(Path=Path,hashlib=hashlib,json=json,os=os,stat=stat,_MAX_FILE=24*1024**2,_MAX_VALUE=512*1024,AcquisitionError=Error,FetchErrorCode=SimpleNamespace(STORAGE='storage',LIMIT='limit',WORKSPACE='workspace'))
    exec(compile(ast.fix_missing_locations(ast.Module(body=[cls],type_ignores=[])),str(source),'exec'),space)
    return space['_PipelineJob'](None,root,root,'fixture')
